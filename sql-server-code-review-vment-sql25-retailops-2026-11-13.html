3. Make the daily sales date filter sargable

USE [RetailOps];
GO

CREATE OR ALTER PROCEDURE dbo.usp_GetDailySales
    @ReportDate date
AS
BEGIN
    SET NOCOUNT ON;

    SELECT p.Category,
           COUNT(DISTINCT o.OrderId) AS Orders,
           SUM(ol.Quantity * ol.UnitPrice) AS Revenue
    FROM dbo.Orders AS o
    JOIN dbo.OrderLines AS ol
        ON ol.OrderId = o.OrderId
    JOIN dbo.Products AS p
        ON p.ProductId = ol.ProductId
    WHERE o.OrderDate >= @ReportDate
      AND o.OrderDate < DATEADD(day, 1, @ReportDate)
      AND o.Status <> 'Cancelled'
    GROUP BY p.Category
    ORDER BY Revenue DESC;
END;
GO

4. Replace cursor-based loyalty recalculation with one set-based update

USE [RetailOps];
GO

CREATE OR ALTER PROCEDURE dbo.usp_RecalcLoyaltyTiers
    @Region varchar(20),
    @AsOf date
AS
BEGIN
    SET NOCOUNT ON;

    ;WITH Spend AS
    (
        SELECT o.CustomerId,
               SUM(ol.Quantity * ol.UnitPrice) AS Spend
        FROM dbo.Customers AS rc
        JOIN dbo.Orders AS o
            ON o.CustomerId = rc.CustomerId
        JOIN dbo.OrderLines AS ol
            ON ol.OrderId = o.OrderId
        WHERE rc.Region = @Region
          AND o.OrderDate >= DATEADD(year, -1, @AsOf)
          AND o.OrderDate < @AsOf
          AND o.Status <> 'Cancelled'
        GROUP BY o.CustomerId
    )
    UPDATE c
       SET LoyaltyTier =
           CASE
               WHEN ISNULL(s.Spend, 0) >= 12500 THEN 'Gold'
               WHEN ISNULL(s.Spend, 0) >= 9500 THEN 'Silver'
               ELSE 'Bronze'
           END
    FROM dbo.Customers AS c
    LEFT JOIN Spend AS s
        ON s.CustomerId = c.CustomerId
    WHERE c.Region = @Region;
END;
GO

5. Serialize concurrent cart-item update/insert operations

USE [RetailOps];
GO

CREATE OR ALTER PROCEDURE dbo.usp_AddCartItem
    @CartId int,
    @ProductId int,
    @Quantity int
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;

    BEGIN TRY
        BEGIN TRANSACTION;

        UPDATE ci WITH (UPDLOCK, HOLDLOCK)
           SET Quantity = Quantity + @Quantity
        FROM dbo.CartItems AS ci
        WHERE ci.CartId = @CartId
          AND ci.ProductId = @ProductId;

        IF @@ROWCOUNT = 0
        BEGIN
            INSERT dbo.CartItems
                (CartId, ProductId, Quantity)
            VALUES
                (@CartId, @ProductId, @Quantity);
        END;

        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH
        IF XACT_STATE() <> 0
            ROLLBACK TRANSACTION;

        THROW;
    END CATCH;
END;
GO
